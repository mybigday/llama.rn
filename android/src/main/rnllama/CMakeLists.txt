# Check if building standalone (for prebuilts) or as subdirectory (for source build)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(RNLLAMA_STANDALONE_BUILD ON)
    message(STATUS "Building rnllama as standalone (for prebuilt distribution)")
else()
    set(RNLLAMA_STANDALONE_BUILD OFF)
    message(STATUS "Building rnllama as subdirectory (will be linked to JNI wrapper)")
endif()

# --- sources shared by every variant ------------------------------------------
# A variant differs from the others only in its CPU flags (-march=...) and in
# the macros below. Two groups of sources can tell the difference and are
# compiled per variant:
#   - the ggml-cpu backend, whose SIMD kernels are selected by the CPU flags
#   - sources that read one of these macros (e.g. backend registration)
# Everything else is compiled once per ABI, at the ABI's baseline, and linked
# into every variant. Upstream llama.cpp scopes its CPU flags to ggml-cpu the
# same way.
#
# Every macro that only some variants define belongs in this list.
set(RNLLAMA_VARIANT_MACROS
    GGML_CPU_GENERIC
    GGML_USE_HEXAGON
    GGML_USE_OPENCL
    GGML_OPENCL_USE_ADRENO_KERNELS
    GGML_OPENCL_EMBED_KERNELS
    GGML_OPENCL_SOA_Q
    GGML_OPENCL_TARGET_VERSION
)
string(REPLACE ";" "|" variant_macro_regex "${RNLLAMA_VARIANT_MACROS}")

set(RNLLAMA_VARIANT_SOURCES ${RNLLAMA_GGML_CPU_SOURCES})
set(RNLLAMA_SHARED_SOURCES ${RNLLAMA_CORE_SOURCES})
list(REMOVE_ITEM RNLLAMA_SHARED_SOURCES ${RNLLAMA_GGML_CPU_SOURCES})
# The split below follows the contents of these sources, so redo it when one changes.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${RNLLAMA_SHARED_SOURCES})
foreach(source ${RNLLAMA_SHARED_SOURCES})
    file(STRINGS ${source} reads_variant_macro REGEX "${variant_macro_regex}" LIMIT_COUNT 1)
    if (reads_variant_macro)
        list(REMOVE_ITEM RNLLAMA_SHARED_SOURCES ${source})
        list(APPEND RNLLAMA_VARIANT_SOURCES ${source})
    endif()
endforeach()

# A shared source could still see one of the macros through a header, and would
# then silently be built without it for the variants that define it. Poisoning
# the names turns that into a compile error ("attempt to use a poisoned
# identifier"); the sources that hit it have to move to RNLLAMA_VARIANT_SOURCES.
string(REPLACE ";" " " poisoned_macros "${RNLLAMA_VARIANT_MACROS}")
set(RNLLAMA_SHARED_SOURCE_GUARD ${CMAKE_CURRENT_BINARY_DIR}/rnllama-shared-source-guard.h)
file(GENERATE OUTPUT ${RNLLAMA_SHARED_SOURCE_GUARD} CONTENT
"// Generated by android/src/main/rnllama/CMakeLists.txt, see RNLLAMA_VARIANT_MACROS there.
#pragma GCC poison ${poisoned_macros}
")

# Flags every object of every variant is compiled with.
# ThinLTO rather than monolithic LTO: linking is parallel and incremental, and
# peak memory stays low (full LTO here OOM-killed 16 GB CI runners).
set(RNLLAMA_COMPILE_OPTIONS
    -DGGML_USE_CPU -DGGML_USE_CPU_REPACK -D_GNU_SOURCE
    -pthread -fvectorize -ffp-model=fast -fno-finite-math-only -flto=thin
    # Optimize for size and performance
    -O3 -DNDEBUG -ffunction-sections -fdata-sections
)
if (CMAKE_BUILD_TYPE AND CMAKE_BUILD_TYPE STREQUAL "Debug")
    list(APPEND RNLLAMA_COMPILE_OPTIONS -DRNLLAMA_ANDROID_ENABLE_LOGGING)
endif ()

# EXCLUDE_FROM_ALL: only built for the variants that are enabled for this ABI.
add_library(rnllama_shared_objects OBJECT EXCLUDE_FROM_ALL ${RNLLAMA_SHARED_SOURCES})
target_compile_options(rnllama_shared_objects PRIVATE
    ${RNLLAMA_COMPILE_OPTIONS} -include ${RNLLAMA_SHARED_SOURCE_GUARD})
set_target_properties(rnllama_shared_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)

# At every link, ThinLTO optimizes and generates code for each module again:
# about a third of a variant's build. lld can cache that per module, which the
# variants after the first then reuse for the shared objects, and rebuilds for
# whatever did not change. RNLLAMA_THINLTO_CACHE_DIR (CMake or environment)
# moves the cache out of the build tree, e.g. to persist it on CI.
#
# lld drops entries that went unused for a week, which a cache restored from an
# archive never looks like, so the size is capped as well: a few times what
# one build of every variant of both ABIs needs.
set(RNLLAMA_THINLTO_CACHE_POLICY cache_size_bytes=4g)
if (NOT RNLLAMA_THINLTO_CACHE_DIR)
    set(RNLLAMA_THINLTO_CACHE_DIR "$ENV{RNLLAMA_THINLTO_CACHE_DIR}")
endif()
if (NOT RNLLAMA_THINLTO_CACHE_DIR)
    set(RNLLAMA_THINLTO_CACHE_DIR "${CMAKE_BINARY_DIR}/thinlto-cache")
endif()
message(STATUS "rnllama: ThinLTO cache at ${RNLLAMA_THINLTO_CACHE_DIR}")

function(build_rnllama_library target_name arch cpu_flags)
    rnllama_variant_enabled(${target_name} VARIANT_ENABLED)
    if (NOT VARIANT_ENABLED)
        message(STATUS "Skipping ${target_name} - not in RNLLAMA_ANDROID_VARIANTS")
        return()
    endif()

    set(ENABLE_OPENCL OFF)
    if (${target_name} MATCHES ".*_opencl$")
        set(ENABLE_OPENCL ON)
    endif ()

    set(ENABLE_HEXAGON OFF)
    if (${target_name} MATCHES ".*_hexagon.*")
        set(ENABLE_HEXAGON ON)
    endif ()

    if (NOT ${arch} STREQUAL "generic")
        set(SOURCE_FILES_ARCH
            ${RNLLAMA_GGML_CPU_ARCH_DIR}/${arch}/quants.c
            ${RNLLAMA_GGML_CPU_ARCH_DIR}/${arch}/repack.cpp
        )
    endif ()

    # What this variant compiles for itself (see RNLLAMA_VARIANT_SOURCES). It is
    # a target of its own so that it never waits for another variant's link.
    set(variant_objects ${target_name}_objects)
    add_library(${variant_objects} OBJECT
        ${RNLLAMA_VARIANT_SOURCES}
        ${SOURCE_FILES_ARCH}
    )
    target_compile_options(${variant_objects} PRIVATE ${RNLLAMA_COMPILE_OPTIONS} ${cpu_flags})
    set_target_properties(${variant_objects} PROPERTIES POSITION_INDEPENDENT_CODE ON)

    if (${arch} STREQUAL "generic")
        target_compile_options(${variant_objects} PRIVATE -DGGML_CPU_GENERIC)
    endif ()

    # Build as shared library for prebuilt distribution
    add_library(
        ${target_name}
        SHARED
        $<TARGET_OBJECTS:rnllama_shared_objects>
        $<TARGET_OBJECTS:${variant_objects}>
    )
    set_target_properties(${target_name} PROPERTIES LINKER_LANGUAGE CXX)

    target_link_libraries(${target_name} PRIVATE ${LOG_LIB} android ${RNLLAMA_VIDEO_LIBS})

    # Hexagon backend support
    # Note: Hexagon requires building DSP libraries with Hexagon SDK toolchain
    # This is a complex process that needs to be done separately
    if (ENABLE_HEXAGON)
        # Hexagon SDK is optional — if not present, build the target without Hexagon support
        set(HEXAGON_SDK_AVAILABLE OFF)
        if (HEXAGON_SDK_ROOT AND HEXAGON_TOOLS_ROOT)
            # Verify libcdsprpc.so exists
            set(CDSPRPC_LIB "${HEXAGON_SDK_ROOT}/ipc/fastrpc/remote/ship/android_aarch64/libcdsprpc.so")
            if (EXISTS ${CDSPRPC_LIB})
                set(HEXAGON_SDK_AVAILABLE ON)
                message(STATUS "Hexagon SDK verified and available")
            else()
                message(WARNING
                    "Hexagon SDK found but libcdsprpc.so is missing at ${CDSPRPC_LIB}. "
                    "Building ${target_name} without Hexagon support.")
            endif()
        else()
            message(STATUS "Hexagon SDK not found — building ${target_name} without Hexagon support.")
        endif()

        if (HEXAGON_SDK_AVAILABLE)
            # Add host-side hexagon sources
            target_sources(${variant_objects} PRIVATE
                ${RNLLAMA_GGML_HEXAGON_DIR}/ggml-hexagon.cpp
                ${RNLLAMA_GGML_HEXAGON_DIR}/htp-kparams.cpp
                ${RNLLAMA_GGML_HEXAGON_DIR}/htp-opnode.cpp
                ${RNLLAMA_GGML_HEXAGON_DIR}/htp-fusion.cpp
                ${RNLLAMA_GGML_HEXAGON_DIR}/htp-drv.cpp
                ${RNLLAMA_GGML_HEXAGON_DIR}/htp-repack.cpp
            )

            # The Android host build consumes generated QAIC artifacts copied by
            # scripts/build-hexagon-htp.sh. Keep the failure mode explicit when
            # they have not been generated yet.
            set(HEXAGON_PREBUILT_HTP_DIR "${RNLLAMA_GGML_HEXAGON_DIR}/htp/v73/")
            set(HEXAGON_HOST_STUB "${HEXAGON_PREBUILT_HTP_DIR}/htp_iface_stub.c")
            set(HEXAGON_HOST_HEADER "${HEXAGON_PREBUILT_HTP_DIR}/htp_iface.h")

            if (NOT EXISTS ${HEXAGON_HOST_STUB} OR NOT EXISTS ${HEXAGON_HOST_HEADER})
                message(FATAL_ERROR
                    "Hexagon host QAIC artifacts are missing. "
                    "Run scripts/build-hexagon-htp.sh to generate ${HEXAGON_PREBUILT_HTP_DIR}.")
            endif()

            target_sources(${variant_objects} PRIVATE
                ${HEXAGON_HOST_STUB}
            )

            target_include_directories(${variant_objects} PRIVATE
                ${HEXAGON_SDK_ROOT}/incs
                ${HEXAGON_SDK_ROOT}/incs/stddef
                ${HEXAGON_SDK_ROOT}/ipc/fastrpc/rpcmem/inc
                ${HEXAGON_SDK_ROOT}/utils/examples
                ${RNLLAMA_GGML_HEXAGON_DIR}
                ${RNLLAMA_GGML_HEXAGON_DIR}/htp
                ${HEXAGON_PREBUILT_HTP_DIR}
                ${CMAKE_CURRENT_BINARY_DIR}
            )

            # Link Hexagon SDK libraries
            # These are stub libraries that communicate with the DSP
            # libcdsprpc.so contains all needed symbols: remote_handle64_*, dspqueue_*, rpcmem_*
            # Note: CDSPRPC_LIB is already verified to exist above
            target_link_libraries(${target_name} PRIVATE ${CDSPRPC_LIB})
            message(STATUS "Linking Hexagon cdsprpc library: ${CDSPRPC_LIB}")

            target_compile_options(${variant_objects} PRIVATE
                -DGGML_USE_HEXAGON
            )

            message(STATUS "Hexagon backend enabled for ${target_name}")
            if (NOT RNLLAMA_STANDALONE_BUILD)
                message(STATUS "  Note: DSP libraries (libggml-htp-*.so) must be built separately")
                message(STATUS "  Run: scripts/build-hexagon-htp.sh to build DSP libraries")
            endif()
        else()
            # Only show warning when building from source
            if (NOT RNLLAMA_STANDALONE_BUILD)
                message(WARNING
                    "Hexagon SDK not found. Hexagon backend will not be built.")
            endif()
        endif()
    endif ()

    # OpenCL only for the special target
    if (ENABLE_OPENCL)
        target_include_directories(${variant_objects} PRIVATE ${RNLLAMA_OPENCL_HEADERS_DIR})

        # Discover kernels from the vendor tree so upstream additions do not leave
        # ggml-opencl.cpp including headers that our Android build never generates.
        set(GGML_OPENCL_KERNEL_DIR ${RNLLAMA_GGML_OPENCL_DIR}/kernels)
        file(GLOB GGML_OPENCL_KERNEL_SOURCES CONFIGURE_DEPENDS
            "${GGML_OPENCL_KERNEL_DIR}/*.cl"
        )
        set(GGML_OPENCL_KERNEL_HEADERS "")

        foreach(input_file ${GGML_OPENCL_KERNEL_SOURCES})
            get_filename_component(kernel ${input_file} NAME_WE)
            set(output_file ${CMAKE_CURRENT_BINARY_DIR}/${kernel}.cl.h)
            message(STATUS "opencl: embedding kernel ${kernel}")
            add_custom_command(
                OUTPUT ${output_file}
                COMMAND ${CMAKE_COMMAND} -E env PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}/..
                        ${Python3_EXECUTABLE} ${GGML_OPENCL_KERNEL_DIR}/embed_kernel.py
                        ${input_file} ${output_file}
                DEPENDS ${input_file} ${GGML_OPENCL_KERNEL_DIR}/embed_kernel.py
                COMMENT "Embedding OpenCL kernel: ${kernel}.cl"
            )

            list(APPEND GGML_OPENCL_KERNEL_HEADERS ${output_file})
        endforeach()

        set(OPENCL_STUB_DIR ${RNLLAMA_ROOT_DIR}/bin/${ANDROID_ABI})
        set(OPENCL_STUB     ${OPENCL_STUB_DIR}/libOpenCL.so)

        if (EXISTS ${OPENCL_STUB})
            # Keep using -lOpenCL so we don't create a packagable .so target.
            target_link_directories(${target_name} PRIVATE ${OPENCL_STUB_DIR})
            target_link_libraries(${target_name} PRIVATE OpenCL)
        else()
            message(WARNING
                "OpenCL library not found. Please build with ./scripts/build-opencl.sh")
        endif()

        target_include_directories(${variant_objects} PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
        target_sources(${variant_objects} PRIVATE
            ${RNLLAMA_GGML_OPENCL_DIR}/ggml-opencl.cpp
            ${RNLLAMA_GGML_OPENCL_DIR}/cl-program-cache.cpp
            ${GGML_OPENCL_KERNEL_HEADERS}
        )
        target_compile_options(${variant_objects} PRIVATE
            -DGGML_USE_OPENCL
            -DGGML_OPENCL_USE_ADRENO_KERNELS
            -DGGML_OPENCL_EMBED_KERNELS
            -DGGML_OPENCL_SOA_Q
            -DGGML_OPENCL_TARGET_VERSION=300
        )
    endif ()

    target_link_options(${target_name} PRIVATE -Wl,--gc-sections)
    # ggml/llama symbols keep upstream names. Bind this library's own references
    # at link time so a second ggml copy in the process (e.g. whisper.rn) can
    # never be picked up via the dynamic linker's global lookup.
    target_link_options(${target_name} PRIVATE -Wl,-Bsymbolic)
    target_link_options(${target_name} PRIVATE -flto=thin
        -Wl,--thinlto-cache-dir=${RNLLAMA_THINLTO_CACHE_DIR}
        -Wl,--thinlto-cache-policy=${RNLLAMA_THINLTO_CACHE_POLICY})

    # Link the variants one after another. Each link already uses every core,
    # so running them side by side only multiplies peak memory, and a variant
    # can reuse the ThinLTO cache entries of the one linked before it.
    get_property(previous_variant GLOBAL PROPERTY RNLLAMA_PREVIOUS_VARIANT)
    if (previous_variant)
        add_dependencies(${target_name} ${previous_variant})
    endif()
    set_property(GLOBAL PROPERTY RNLLAMA_PREVIOUS_VARIANT ${target_name})
endfunction()

# Default target (no specific CPU features)
build_rnllama_library("rnllama" "generic" "")

if (ANDROID_ABI AND ANDROID_ABI STREQUAL "arm64-v8a")
    # ARM64 targets
    build_rnllama_library("rnllama_v8" "arm" "-march=armv8-a")
    build_rnllama_library("rnllama_v8_2" "arm" "-march=armv8.2-a")
    build_rnllama_library("rnllama_v8_2_dotprod" "arm" "-march=armv8.2-a+dotprod")
    build_rnllama_library("rnllama_v8_2_i8mm" "arm" "-march=armv8.2-a+i8mm")
    build_rnllama_library("rnllama_v8_2_dotprod_i8mm" "arm" "-march=armv8.2-a+dotprod+i8mm")
    build_rnllama_library("rnllama_v8_2_dotprod_i8mm_hexagon_opencl" "arm" "-march=armv8.2-a+dotprod+i8mm")

elseif (ANDROID_ABI AND ANDROID_ABI STREQUAL "x86_64")
    # x86_64 target
    build_rnllama_library("rnllama_x86_64" "x86" "-march=x86-64;-mtune=generic;-msse4.2;-mpopcnt")

endif ()
