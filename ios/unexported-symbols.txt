# Symbols that rnllama.framework must not export.
#
# ggml/gguf keep their upstream names (no LM_ prefix). Another ggml-based
# framework in the same app (e.g. whisper.rn) exports the same names, and the
# app-level link binds any reference to whichever framework comes first, so
# nothing outside this framework may resolve ggml through it. Everything the
# JSI glue needs goes through rnllama:: wrappers (see rn-llama.h).
#
# ld64 glob syntax: * ? [abc]. Mangled C++ names encode the identifier length
# before the identifier, hence the one/two/three-digit variants.

# ggml / gguf C API and C internals
_ggml_*
_gguf_*

_dequantize_*
_iq2xs_*
_iq3xs_*

__ZZN4ggml*
